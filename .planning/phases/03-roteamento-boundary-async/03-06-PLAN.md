package br.com.erpkit.whatsapp.config;

import org.springframework.boot.test.context.TestConfiguration;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Primary;
import org.springframework.core.task.SyncTaskExecutor;
import org.springframework.core.task.TaskExecutor;

/**
 * Substitui o bean {@code whatsappTaskExecutor} de producao em testes selecionados.
 *
 * <p>SyncTaskExecutor executa {@code task.run()} inline na thread chamadora —
 * listener {@code @Async} vira sincrono em test. Combinado com
 * {@code @TransactionalEventListener(AFTER_COMMIT)}, o listener dispara apos
 * commit do save() do MensagemService — quando o metodo {@code processarWebhook}
 * retorna, todo o flow ja rodou (sync) — assertions DB seguras sem Awaitility.
 *
 * <p><b>NAO usar em SC-1 timing test</b> — esse precisa de executor real para
 * provar que o ack 200 retorna em <1s mesmo com ERP delay 10s.
 *
 * <p>Uso:
 * <pre>
 * &#64;SpringBootTest
 * &#64;Import(AsyncTestConfig.class)
 * class MyIntegrationTest { ... }
 * </pre>
 */
@TestConfiguration
public class AsyncTestConfig {

    @Bean(name = "whatsappTaskExecutor")
    @Primary
    public TaskExecutor whatsappTaskExecutor() {
        return new SyncTaskExecutor();
    }
}

import com.fasterxml.jackson.databind.ObjectMapper;
import com.github.tomakehurst.wiremock.WireMockServer;
import com.github.tomakehurst.wiremock.core.WireMockConfiguration;
import org.junit.jupiter.api.AfterAll;
import org.junit.jupiter.api.BeforeAll;

import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest(classes = WhatsAppApplication.class)
@AutoConfigureMockMvc
@ActiveProfiles("test")
@Import(AsyncTestConfig.class)
class WebhookAsyncIntegrationTest {

    private static WireMockServer wireMockErp;
    private static WireMockServer wireMockMeta;

    @BeforeAll
    static void startWireMock() {
        wireMockErp = new WireMockServer(WireMockConfiguration.options().dynamicPort());
        wireMockErp.start();
        wireMockMeta = new WireMockServer(WireMockConfiguration.options().dynamicPort());
        wireMockMeta.start();
    }

    @AfterAll
    static void stopWireMock() {
        wireMockErp.stop();
        wireMockMeta.stop();

    @Autowired MockMvc mockMvc;
    @Autowired ObjectMapper objectMapper;

    @BeforeEach
    void resetWireMock() {
        wireMockErp.resetAll();
        wireMockMeta.resetAll();
    }

    private byte[] fixture(String nome) throws Exception {
        try (InputStream in = new ClassPathResource("fixtures/webhook/" + nome).getInputStream()) {
            return StreamUtils.copyToByteArray(in);
        }
    }

    private String computeSignature(byte[] body, String appSecret) throws Exception {
        Mac mac = Mac.getInstance("HmacSHA256");
        mac.init(new SecretKeySpec(appSecret.getBytes(StandardCharsets.UTF_8), "HmacSHA256"));
        return HexFormat.of().formatHex(mac.doFinal(body));
    }

    @Test
    @DisplayName("SC-2: callback payload tem telefone+comando+payload+idCliente formatados corretamente")
    void sc2_callback_payload_correto() throws Exception {
        wireMockErp.stubFor(post(urlEqualTo("/api/modulos/whatsapp/comando"))
            .willReturn(aResponse().withStatus(200)));

        byte[] body = fixture("text-portugues.json");
        mockMvc.perform(post("/webhook/whatsapp")
                .header("X-Hub-Signature-256", "sha256=" + computeSignature(body, "test-app-secret"))
                .contentType(MediaType.APPLICATION_JSON)
                .content(body))
            .andExpect(status().isOk());

        // Capturar request recebido pelo WireMock e validar payload JSON
        var requests = wireMockErp.findAll(postRequestedFor(urlEqualTo("/api/modulos/whatsapp/comando")));
        assertThat(requests).hasSize(1);
        JsonNode payload = objectMapper.readTree(requests.get(0).getBodyAsString());
        assertThat(payload.get("telefone").asText()).isEqualTo("554784178525");
        // primeira palavra do conteudo "Olá, gostaria de um orçamento" -> "olá,"
        // (ComandoExtractor lowercase + split por whitespace; nao remove pontuacao)
        // Para test estavel, fixture poderia ser ajustada; aceitar valor real do parser.
        assertThat(payload.get("comando").asText()).isNotBlank();
        assertThat(payload.get("payload").asText()).contains("Ol");

    @Test
    @DisplayName("SC-3: ERP callback 5xx persistente — webhook retorna 200 + log.error capturado, sem retentativas alem do Resilience4j")
    void sc3_callback_5xx_nao_trava_webhook() throws Exception {
        wireMockErp.stubFor(post(urlEqualTo("/api/modulos/whatsapp/comando"))
            .willReturn(aResponse().withStatus(500)));

        byte[] body = fixture("text-portugues.json");

        // Webhook RETORNA 200 mesmo com ERP 5xx (ack-first preservado)
        mockMvc.perform(post("/webhook/whatsapp")
                .header("X-Hub-Signature-256", "sha256=" + computeSignature(body, "test-app-secret"))
                .contentType(MediaType.APPLICATION_JSON)
                .content(body))
            .andExpect(status().isOk());

        // ERP recebeu 3 tentativas (Resilience4j max-attempts=3) e parou — sem retry adicional
        // (D-08, ROU-03). Apos 3 falhas, fallback engole.
        int erpCount = wireMockErp.findAll(postRequestedFor(urlEqualTo("/api/modulos/whatsapp/comando"))).size();
        assertThat(erpCount).as("Resilience4j max-attempts=3 — sem retentativas adicionais").isEqualTo(3);
    }

    @DisplayName("SC-4: media download e a primeira acao async — Meta GET ANTERIOR ao ERP POST")
    void sc4_media_download_primeira_acao() throws Exception {
        // Stub Meta Graph: step 1 (metadata) + step 2 (bytes)
        wireMockMeta.stubFor(get(urlMatching("/MEDIA-DOC-001"))
            .willReturn(aResponse().withStatus(200).withHeader("Content-Type", "application/json")
                .withBody("""
                    {"url":"%s/cdn/MEDIA-DOC-001/data","mime_type":"application/pdf","filename":"fatura.pdf"}
                    """.formatted(wireMockMeta.baseUrl()))));
        wireMockMeta.stubFor(get(urlEqualTo("/cdn/MEDIA-DOC-001/data"))
            .willReturn(aResponse().withStatus(200).withBody(new byte[]{1,2,3,4,5})));
        wireMockErp.stubFor(post(urlEqualTo("/api/modulos/whatsapp/comando"))
            .willReturn(aResponse().withStatus(200)));

        byte[] body = fixture("document-pdf.json"); // fixture que tem mediaId="MEDIA-DOC-001"

        mockMvc.perform(post("/webhook/whatsapp")
                .header("X-Hub-Signature-256", "sha256=" + computeSignature(body, "test-app-secret"))

                .content(body))
            .andExpect(status().isOk());

        assertThat(metaEvents).isNotEmpty();

package br.com.erpkit.whatsapp.controller;

import br.com.erpkit.whatsapp.WhatsAppApplication;
import com.github.tomakehurst.wiremock.WireMockServer;
import com.github.tomakehurst.wiremock.core.WireMockConfiguration;
// ... mesmos imports da Classe A, EXCETO @Import(AsyncTestConfig.class) e SyncTaskExecutor
import org.awaitility.Awaitility;

import java.time.Duration;

import static com.github.tomakehurst.wiremock.client.WireMock.*;
import static org.assertj.core.api.Assertions.assertThat;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest(classes = WhatsAppApplication.class)
@AutoConfigureMockMvc
@ActiveProfiles("test")
// SEM @Import(AsyncTestConfig.class) — usa pool real para validar timing
class WebhookAsyncTimingIntegrationTest {

    // ... mesmo setup WireMock + helpers da Classe A

    @Test
    @DisplayName("SC-1: ack 200 em <1s mesmo com ERP delay 10s (pool real, listener async)")
    void sc1_ack_first_under_1s() throws Exception {
        wireMockErp.stubFor(post(urlEqualTo("/api/modulos/whatsapp/comando"))
            .willReturn(aResponse().withStatus(200).withFixedDelay(10000)));

        byte[] body = fixture("text-portugues.json");
        long start = System.currentTimeMillis();
        mockMvc.perform(post("/webhook/whatsapp")
                .header("X-Hub-Signature-256", "sha256=" + computeSignature(body, "test-app-secret"))
                .contentType(MediaType.APPLICATION_JSON)
                .content(body))
            .andExpect(status().isOk());
        long elapsed = System.currentTimeMillis() - start;

        assertThat(elapsed)
            .as("Ack-first: 200 ao Meta deve retornar em <1s mesmo com ERP delay 10s")
            .isLessThan(1000);

        // Awaitility: callback eventualmente chegou (apos delay 10s).
        // BUT: Resilience4j callbackTimeout=500ms (test profile), entao o ERP chamada
        // vai dar TIMEOUT antes de receber resposta — listener fallback log.error.
        // Para SC-1 puro (timing do ack), o que importa e o elapsed < 1000ms.
        // Para confirmar listener rodou, esperar ate 12s por ANY request ao WireMock.
        Awaitility.await().atMost(Duration.ofSeconds(12)).untilAsserted(() ->
            assertThat(wireMockErp.findAll(postRequestedFor(urlEqualTo("/api/modulos/whatsapp/comando"))))
                .as("Listener async eventualmente chamou ERP callback")
                .isNotEmpty()
        );
    }
}
